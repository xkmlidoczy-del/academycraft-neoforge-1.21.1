def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def main():
    witnesses=[]
    for rel in COMPILE_WITNESSES + READ_ONLY_WITNESSES:
        source=REFERENCE/rel
        target=ORACLE/('original' if rel in COMPILE_WITNESSES else 'witnesses')/rel
        target.parent.mkdir(parents=True, exist_ok=True)
        shutil.copyfile(source,target)
        assert digest(source)==digest(target)
        witnesses.append({'canonical':str(source.relative_to(ROOT)),
                          'copied':str(target.relative_to(STAGE)),
                          'sha256':digest(source),
                          'compiled':rel in COMPILE_WITNESSES})
    for rel,content in HOSTS.items():
        target=ORACLE/'hosts'/rel
        target.parent.mkdir(parents=True,exist_ok=True)
        target.write_text(content)
    (ORACLE/'source-hashes.json').write_text(json.dumps(witnesses,indent=2)+'\n')
    print(f'Copied {len(witnesses)} byte-identical witnesses; declared {len(HOSTS)} hosts')

if __name__ == '__main__':
    main()
