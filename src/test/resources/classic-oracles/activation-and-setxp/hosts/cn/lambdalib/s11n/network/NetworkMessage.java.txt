/* Test-only synchronous message dispatch; business setters/listener execute unchanged originals. */
package cn.lambdalib.s11n.network;
public class NetworkMessage {
 @java.lang.annotation.Retention(java.lang.annotation.RetentionPolicy.RUNTIME)
 public @interface Listener { String channel(); cpw.mods.fml.relauncher.Side[] side(); }
 public static final java.util.List<String> messages=new java.util.ArrayList<>();
 public static void sendToServer(Object instance,String channel,Object...params){messages.add("server:"+channel+":"+java.util.Arrays.toString(params));}
 public static void sendTo(Object player,Object instance,String channel,Object...params){messages.add("remote:"+channel+":"+java.util.Arrays.toString(params));}
 public static void sendToSelf(Object instance,String channel,Object...params){
  messages.add("self:"+channel+":"+java.util.Arrays.toString(params));
  for(var method:instance.getClass().getDeclaredMethods()){
   var listener=method.getAnnotation(Listener.class);if(listener==null||!listener.channel().equals(channel))continue;
   if(java.util.Arrays.stream(listener.side()).noneMatch(side->side==cpw.mods.fml.relauncher.Side.SERVER))continue;
   try{method.setAccessible(true);method.invoke(instance,params);}catch(java.lang.reflect.InvocationTargetException e){throw new AssertionError(e.getCause());}catch(ReflectiveOperationException e){throw new AssertionError(e);}
  }
 }
}
